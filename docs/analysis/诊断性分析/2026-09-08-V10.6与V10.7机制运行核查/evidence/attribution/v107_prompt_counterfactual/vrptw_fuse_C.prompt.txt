Idea: The algorithm computes a composite insertion cost for each feasible candidate, prioritizing spatial proximity while strongly penalizing temporal inefficiency. It calculates raw distance and waiting time, normalizing them by the maximum values among current candidates to ensure scale-invariance. To prevent route failure, it incorporates a dynamic regret term that exponentially weights the tightness of the remaining time window (slack). Nodes with minimal slack are heavily penalized if not selected immediately, balancing short-term distance optimization with long-term route feasibility.

import numpy as np

def select_next_node(current_node: int, depot: int, unvisited_nodes: np.ndarray, rest_capacity: float, current_time: float, demands: np.ndarray, distance_matrix: np.ndarray, time_windows: np.ndarray) -> int:
    """
    Selects the next node using a normalized composite score of distance, waiting time, and dynamic time-window regret.
    
    The score is defined as:
    Score = w_dist * norm_dist + w_wait * norm_wait + w_regret * (1 - norm_slack)
    
    Where:
    - norm_dist is the distance normalized by the max distance among feasible candidates.
    - norm_wait is the waiting time normalized by the max wait among feasible candidates.
    - norm_slack is the remaining slack normalized by the max slack among feasible candidates.
    - A higher w_regret ensures that nodes with tight time windows (low slack) are prioritized to avoid infeasibility later.
    """
    if len(unvisited_nodes) == 0:
        return depot

    unvisited_nodes = np.asarray(unvisited_nodes)
    
    if len(unvisited_nodes) == 1:
        node = int(unvisited_nodes[0])
        # Safety check: ensure the single node is actually feasible
        if demands[node] <= rest_capacity:
            arrival = current_time + distance_matrix[current_node, node]
            if arrival <= time_windows[node, 1]:
                return node
        return depot

    # Weights tuned to balance proximity and time feasibility
    w_dist = 1.0
    w_wait = 0.8
    w_regret = 1.5

    candidates = []
    max_dist = 0.0
    max_wait = 0.0
    max_slack = 0.0

    for node in unvisited_nodes:
        node_idx = int(node)
        
        # 1. Capacity Check
        if demands[node_idx] > rest_capacity:
            continue
            
        # 2. Distance and Arrival Time
        dist = distance_matrix[current_node, node_idx]
        arrival_time = current_time + dist
        
        # 3. Time Window Check
        tw_start = time_windows[node_idx, 0]
        tw_end = time_windows[node_idx, 1]
        
        if arrival_time > tw_end:
            continue
            
        # 4. Metrics Calculation
        wait_time = max(0.0, tw_start - arrival_time)
        service_start = max(arrival_time, tw_start)
        slack = tw_end - service_start
        
        # Update normalization bounds
        if dist > max_dist: max_dist = dist
        if wait_time > max_wait: max_wait = wait_time
        if slack > max_slack: max_slack = slack
        
        candidates.append({
            'node': node_idx,
            'dist': dist,
            'wait': wait_time,
            'slack': slack
        })

    if not candidates:
        return depot

    if len(candidates) == 1:
        return candidates[0]['node']

    # Avoid division by zero
    if max_dist == 0: max_dist = 1.0
    if max_wait == 0: max_wait = 1.0
    if max_slack == 0: max_slack = 1.0

    best_score = float('inf')
    best_node = depot

    for c in candidates:
        # Normalize components to [0, 1]
        norm_dist = c['dist'] / max_dist
        norm_wait = c['wait'] / max_wait
        norm_slack = c['slack'] / max_slack
        
        # Regret Term: (1 - norm_slack)
        # If slack is max, regret is 0 (no penalty).
        # If slack is 0, regret is 1 (max penalty).
        # This encourages visiting nodes that are becoming infeasible sooner.
        regret = 1.0 - norm_slack
        
        # Composite Score
        score = w_dist * norm_dist + w_wait * norm_wait + w_regret * regret
        
        if score < best_score:
            best_score = score
            best_node = c['node']

    return best_node

It computes immediate travel costs including waiting time, then estimates future feasibility by checking the cheapest valid next step from each candidate (including return to depot). Urgency is derived from the remaining time slack within each node's window using an exponential decay function. A 2-regret term measures the cost difference between the best and second-best immediate alternatives to prevent suboptimal early choices. Dynamic weights balance these components based on overall time-window tightness, penalizing candidates that lead to high future costs or leave little slack for critical nodes.

import numpy as np

def select_next_node(current_node: int, depot: int, unvisited_nodes: np.ndarray, rest_capacity: float, current_time: float, demands: np.ndarray, distance_matrix: np.ndarray, time_windows: np.ndarray) -> int:

            if arrival <= time_windows[node, 1] + 1e-9:
                return node
        return depot

    candidates = []

    for node in unvisited_nodes:
        node_idx = int(node)
        
        # Capacity Check
        if demands[node_idx] > rest_capacity:
            continue

        arrival_time = current_time + dist
        
        tw_start = time_windows[node_idx, 0]
        tw_end = time_windows[node_idx, 1]
        
        # Time Window Feasibility Check
        if arrival_time > tw_end + 1e-9:
            continue
        
        wait_time = max(0.0, tw_start - arrival_time)
        service_start = max(arrival_time, tw_start)
        slack = tw_end - service_start
        
        # Base Cost: Distance + Waiting Time
        base_cost = dist + 0.3 * wait_time

        candidates.append({
            'node': node_idx,
            'cost': base_cost,
            'slack': slack,
            'arrival_time': service_start,
            'remaining_capacity': rest_capacity - demands[node_idx]
        })
    
    if not candidates:
        return depot
        
    if len(candidates) == 1:
        return candidates[0]['node']

    n_cand = len(candidates)
    costs = np.array([c['cost'] for c in candidates])
    slacks = np.array([c['slack'] for c in candidates])
    nodes = np.array([c['node'] for c in candidates])
    arrivals = np.array([c['arrival_time'] for c in candidates])
    rem_caps = np.array([c['remaining_capacity'] for c in candidates])

    # Normalize costs to [0, 1]
    min_c, max_c = np.min(costs), np.max(costs)
    cost_range = max_c - min_c if max_c > min_c + 1e-9 else 1.0
    norm_costs = (costs - min_c) / cost_range

    # Calculate Adaptive Urgency
    avg_slack = np.mean(slacks)
    min_slack = np.min(slacks)
    # Dynamic lambda: tighter windows -> steeper decay
    lambda_urgency = 0.1 / (1.0 + avg_slack / 50.0)
    urgencies = np.exp(-lambda_urgency * slacks)
    
    # Additional urgency boost if min_slack is very small
    if min_slack < 5:
        urgencies = 0.5 * urgencies + 0.5 * (1.0 / (1.0 + min_slack))

    # Calculate Regret (2-regret)
    # Regret is the cost of losing the best alternative if we don't pick this one.
    regrets = np.zeros(n_cand)
    for i in range(n_cand):
        if n_cand > 2:
            mask = np.ones(n_cand, dtype=bool)
            mask[i] = False
            best_alt_cost = np.min(costs[mask])
        else:
            other_idx = 1 - i
            best_alt_cost = costs[other_idx]
        regrets[i] = best_alt_cost - costs[i]
    
    min_reg, max_reg = np.min(regrets), np.max(regrets)
    regret_range = max_reg - min_reg if max_reg > min_reg + 1e-9 else 1.0
    norm_regrets = (regrets - min_reg) / regret_range

    # Calculate Lookahead Cost: 2-step simulation
    # For each candidate, estimate the minimum cost to the next step (another customer or depot)
    lookahead_costs = np.zeros(n_cand)
    for i in range(n_cand):
        current_node_idx = nodes[i]
        current_time_after_visit = arrivals[i]
        current_cap_after_visit = rem_caps[i]
        
        min_next_cost = float('inf')
        
        # Option 1: Go to Depot
        dist_to_depot = distance_matrix[current_node_idx, depot]
        min_next_cost = min(min_next_cost, dist_to_depot)
        
        # Option 2: Go to another feasible customer
        for j in range(n_cand):
            if i == j:
                continue
            
            node_j = nodes[j]
            
            # Check capacity
            if demands[node_j] > current_cap_after_visit:
                continue
                
            dist_ij = distance_matrix[current_node_idx, node_j]
            arrival_j = current_time_after_visit + dist_ij
            
            tw_start_j = time_windows[node_j, 0]
            tw_end_j = time_windows[node_j, 1]
            
            if arrival_j > tw_end_j + 1e-9:
                continue
            
            wait_j = max(0.0, tw_start_j - arrival_j)
            service_start_j = max(arrival_j, tw_start_j)
            cost_j = dist_ij + 0.3 * wait_j
            
            if cost_j < min_next_cost:
                min_next_cost = cost_j
        
        # If no other customer is feasible, min_next_cost remains dist_to_depot
        if min_next_cost == float('inf'):
            min_next_cost = 0.0
            
        lookahead_costs[i] = min_next_cost

    # Normalize lookahead costs
    min_lk, max_lk = np.min(lookahead_costs), np.max(lookahead_costs)
    lk_range = max_lk - min_lk if max_lk > min_lk + 1e-9 else 1.0
    norm_lookaheads = (lookahead_costs - min_lk) / lk_range

    # Adaptive Weights
    # Base weights balancing the four components
    w_cost = 0.4
    w_urgency = 0.3
    w_regret = 0.15
    w_lookahead = 0.15
    
    # Adapt weights based on slack tightness
    # If time windows are tight, prioritize urgency and regret to avoid getting stuck
    if min_slack < 10:
        w_urgency *= 2.0
        w_regret *= 1.5
        w_cost *= 0.8
        w_lookahead *= 0.5
    elif avg_slack < 20:
        w_urgency *= 1.5

    total_w = w_cost + w_urgency + w_regret + w_lookahead
    w_cost /= total_w
    w_urgency /= total_w
    w_regret /= total_w
    w_lookahead /= total_w

    # Final Score Minimization
    # 1. Cost: Lower is better
    # 2. Urgency: Higher urgency is better -> Penalize low urgency (1 - urgency)
    # 3. Regret: Higher regret is better -> Penalize low regret (1 - regret)
    # 4. Lookahead: Lower distance to next step is better
    
    score = (
        w_cost * norm_costs + 
        w_urgency * (1.0 - urgencies) + 
        w_regret * (1.0 - norm_regrets) +
        w_lookahead * norm_lookaheads
    )
    
    best_idx = np.argmin(score)
    return int(nodes[best_idx])

    2-step lookahead, and adaptive urgency.

        current_node: ID of the current node.

        unvisited_nodes: Array of feasible unvisited node IDs.

        distance_matrix: Pairwise distance matrix.

            'remaining_capacity': rest_capacity - demands[node_idx],
            'demand': demands[node_idx]
        })

        return depot
        
    if len(candidates) == 1:
        return candidates[0]['node']

    costs = np.array([c['cost'] for c in candidates])
    slacks = np.array([c['slack'] for c in candidates])
    nodes = np.array([c['node'] for c in candidates])
    arrivals = np.array([c['arrival_time'] for c in candidates])
    rem_caps = np.array([c['remaining_capacity'] for c in candidates])
    demands_cand = np.array([c['demand'] for c in candidates])

    cost_range = max_c - min_c if max_c > min_c + 1e-9 else 1.0

    # Regret_i = (Best Alternative Cost) - (Cost_i) + (Capacity Penalty if skipping i leads to dead end)
    # We define a capacity risk penalty for each candidate based on how much capacity is left
    # and whether that capacity can serve any remaining high-demand unvisited nodes.
