    Select the next node using exhaustive search for small sets (n <= 8)
    and a 2-step lookahead with greedy completion for larger sets.
    """
    if len(unvisited_nodes) == 0:
        return current_node
        
    if len(unvisited_nodes) == 1:
        return int(unvisited_nodes[0])

    n = len(unvisited_nodes)
    candidates = unvisited_nodes

    if n <= 8:
        # Exhaustive search: compute exact minimum completion cost for each candidate
        best_score = float('inf')
        best_node = candidates[0]
        
        for v in candidates:
            cost_to_v = distance_matrix[current_node, v]
            remaining = [u for u in candidates if u != v]
            
            def dfs(curr, unvisited):
                if not unvisited:
                    return distance_matrix[curr, destination_node]
                min_cost = float('inf')
                for u in unvisited:
                    next_cost = distance_matrix[curr, u] + dfs(u, [x for x in unvisited if x != u])
                    if next_cost < min_cost:
                        min_cost = next_cost
                return min_cost
            
            total_cost = cost_to_v + dfs(v, remaining)
            if total_cost < best_score:
                best_score = total_cost
                best_node = v
        
        return int(best_node)
    else:
        # 2-step lookahead: for each v, pick best 2nd step u, then greedy rest
        best_score = float('inf')
        best_node = candidates[0]
        
        for v in candidates:
            cost_to_v = distance_matrix[current_node, v]
            remaining = [u for u in candidates if u != v]
            
            if not remaining:
                total_cost = cost_to_v + distance_matrix[v, destination_node]
            else:
                # Find best second step u from v
                # We want to minimize d(v, u) + est_cost(u, remaining\{u}, dest)
                # But computing est_cost for all u is expensive.
                # Approximation: Pick u that minimizes d(v, u) + d(u, dest) or similar?
                # Better: Try the 2-3 nearest neighbors of v as u candidates.
                
                # Sort remaining by distance from v
                dist_from_v = distance_matrix[v, remaining]
                sorted_rem_indices = np.argsort(dist_from_v)
                
                # Consider top 2 candidates for the second step
                max_u_candidates = min(2, len(remaining))
                min_est_cost = float('inf')
                
                for idx in range(max_u_candidates):
                    u = remaining[sorted_rem_indices[idx]]
                    cost_v_to_u = dist_from_v[sorted_rem_indices[idx]]
                    rem_after_u = [w for w in remaining if w != u]
                    
                    # Greedy completion from u
                    curr = u
                    rem_cost = 0.0
                    rem_set = set(rem_after_u)
                    
                    while rem_set:
                        d_curr_rem = distance_matrix[curr, list(rem_set)]
                        best_next_idx = np.argmin(d_curr_rem)
                        best_next = list(rem_set)[best_next_idx]
                        rem_cost += d_curr_rem[best_next_idx]
                        rem_set.remove(best_next)
                        curr = best_next
                    
                    rem_cost += distance_matrix[curr, destination_node]
                    total_est = cost_to_v + cost_v_to_u + rem_cost
                    
                    if total_est < min_est_cost:
                        min_est_cost = total_est
                
                total_cost = min_est_cost
            
            if total_cost < best_score:
                best_score = total_cost
                best_node = v
        
        return int(best_node)

    immediate distance to unvisited nodes with the distance to the destination.
    """
    if len(unvisited_nodes) == 0:
        return current_node
        
    if len(unvisited_nodes) == 1:
        return int(unvisited_nodes[0])

    n_unvisited = len(unvisited_nodes)
    
    # Adaptive weight: 0.0 when n >= 15, 0.6 when n <= 2
    # Linear interpolation between these points to balance exploration and exploitation
    if n_unvisited >= 15:
        w = 0.0
    elif n_unvisited <= 2:
        w = 0.6
    else:
        w = 0.6 * (15.0 - n_unvisited) / 13.0

    candidates = unvisited_nodes
    
    # Vectorized distance calculation from current node
    d_cur_to_v = distance_matrix[current_node, candidates]
    
    # Calculate regret term for each candidate
    # Regret: If we don't go to v, we likely go to the best alternative u.
    # The cost of not choosing v is approximated by the distance from that alternative u to the destination.
    # This penalizes choices where the alternative is far from the destination, effectively preferring
    # nodes that are not just close to current, but also lead to a good position for the rest of the tour.
    
    n = len(candidates)
    regret_costs = np.zeros(n)
    
    if n == 2:
        # Special case: best alternative to v is the other node
        # regret_costs[0] = d(node1, dest)
        # regret_costs[1] = d(node0, dest)
        regret_costs[0] = distance_matrix[candidates[1], destination_node]
        regret_costs[1] = distance_matrix[candidates[0], destination_node]
    else:
        # Find indices of two smallest distances
        sorted_indices = np.argsort(d_cur_to_v)
        best_idx_global = sorted_indices[0]
        second_best_idx_global = sorted_indices[1]
        
        best_node_global = candidates[best_idx_global]
        second_best_node_global = candidates[second_best_idx_global]
        
        # For all v != best_node_global, the best alternative is best_node_global.
        # So regret_costs for these v is d(best_node_global, dest).
        # For v == best_node_global, the best alternative is second_best_node_global.
        # So regret_costs for this v is d(second_best_node_global, dest).
        
        base_regret = distance_matrix[best_node_global, destination_node]
        exception_regret = distance_matrix[second_best_node_global, destination_node]
        
        regret_costs[:] = base_regret
        regret_costs[best_idx_global] = exception_regret

    # Score = d_cur_to_v + w * regret_costs
    scores = d_cur_to_v + w * regret_costs
    
    best_idx = np.argmin(scores)
    return int(candidates[best_idx])

    Select the next node using a tiered strategy:
    1. Exhaustive search for small sets (<= 4).
    2. 2-step lookahead with greedy estimation for medium sets (5-20).
    3. Nearest neighbor for large sets (> 20).
    """

        return current_node
        
    if len(unvisited_nodes) == 1:
        return int(unvisited_nodes[0])

    n = len(unvisited_nodes)
    candidates = unvisited_nodes

    if n <= 4:
        # Exhaustive search for optimal completion
        best_score = float('inf')
        best_node = candidates[0]
        
        for v in candidates:
            cost = distance_matrix[current_node, v]
            remaining = [u for u in candidates if u != v]
            
            if not remaining:
                total_cost = cost + distance_matrix[v, destination_node]
            else:
                # Recursive helper for small sets
                def dfs(curr, unvisited):
                    if not unvisited:
                        return distance_matrix[curr, destination_node]

                    for u in unvisited:
                        next_cost = distance_matrix[curr, u] + dfs(u, [x for x in unvisited if x != u])
                        if next_cost < min_cost:
                            min_cost = next_cost
                    return min_cost

                best_node = v

    Select the next node using a 2-step lookahead with 2-opt refinement for
    medium-sized sets (n <= 20), and a weighted distance heuristic for larger sets.
    """

        return int(unvisited_nodes[0])

    candidates = unvisited_nodes

    if n <= 20:
        # 2-step lookahead: for each candidate v, try v->u as the first two moves,
        # then complete the rest using nearest-neighbor + 2-opt.
        best_score = float('inf')

            # 1. Nearest unvisited node from v

                u = remaining_after_v[0]

def _nearest_neighbor_path(start: int, unvisited: list, dist: np.ndarray, dest: int) -> list:
    """Build a nearest-neighbor path from start through all unvisited nodes to dest."""
    path = [start]
    rem = list(unvisited)
    curr = start
    while rem:
        d = dist[curr, rem]
        nxt = rem[np.argmin(d)]
        path.append(nxt)
        rem.remove(nxt)
        curr = nxt
    path.append(dest)
    return path


def _two_opt(path: list, dist: np.ndarray) -> list:
    """Apply 2-opt optimization to a path (including start and end)."""
    n = len(path)
    if n < 3:
        return path
    improved = True
    while improved:
        improved = False
        for i in range(1, n - 2):
            for j in range(i + 1, n - 1):
                # Try reversing path[i+1..j]
                if i == j:
                    continue
                # Old edges: path[i]->path[i+1] and path[j]->path[j+1]
                # New edges: path[i]->path[j] and path[i+1]->path[j+1]
                old_cost = dist[path[i], path[i+1]] + dist[path[j], path[j+1]]
                new_cost = dist[path[i], path[j]] + dist[path[i+1], path[j+1]]
                if new_cost < old_cost - 1e-9:
                    # Reverse the segment
                    path = path[:i+1] + path[i+1:j+1][::-1] + path[j+1:]
                    improved = True
                    break
            if improved:
                break
    return path
