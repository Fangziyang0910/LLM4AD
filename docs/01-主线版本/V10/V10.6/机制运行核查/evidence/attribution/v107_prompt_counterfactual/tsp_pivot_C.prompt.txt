```python
import numpy as np

def select_next_node(current_node: int, destination_node: int, unvisited_nodes: np.ndarray, distance_matrix: np.ndarray) -> int:
    if len(unvisited_nodes) == 0:
        return destination_node

    n_remaining = len(unvisited_nodes)

    if n_remaining == 1:
        return int(unvisited_nodes[0])

    n_total = distance_matrix.shape[0]
    if n_total <= 1:
        return destination_node

    d_curr = distance_matrix[current_node, unvisited_nodes]
    d_dest = distance_matrix[unvisited_nodes, destination_node]

    sub = distance_matrix[np.ix_(unvisited_nodes, unvisited_nodes)]
    np.fill_diagonal(sub, np.inf)

    avg_cohesion = np.sum(sub, axis=1) / (n_remaining - 1)

    min_exit = np.minimum(np.min(sub, axis=1), d_dest)

    if n_remaining > 2:
        sorted_dists = np.sort(sub, axis=1)
        second_nearest = sorted_dists[:, 1]
    else:
        second_nearest = np.full(n_remaining, np.inf)

    scale = np.mean(d_curr)
    if scale < 1e-9:
        scale = 1.0

    progress = (n_total - n_remaining) / (n_total - 1)

    return_weight = 0.1 + 0.6 * (progress * progress * (3 - 2 * progress))

    cohesion_weight = 0.4 * (1 - progress)
    exit_weight = 0.2 * (1 - progress)
    orphan_weight = 0.15 * (1 - progress)

    d_curr_n = d_curr / scale
    d_dest_n = d_dest / scale
    cohesion_n = avg_cohesion / scale
    exit_n = min_exit / scale
    orphan_n = second_nearest / scale

    scores = d_curr_n + return_weight * d_dest_n + cohesion_weight * cohesion_n + exit_weight * exit_n + orphan_weight * orphan_n

    k = min(3, n_remaining - 1)
    flexibility = np.sum(np.sort(sub, axis=1)[:, :k], axis=1)

    tie_breaker = 1e-6 * flexibility / scale

    final_scores = scores + tie_breaker

    if np.any(np.isnan(final_scores)) or np.any(np.isinf(final_scores)):
        final_scores = d_curr_n + 1e-6 * np.arange(n_remaining)

    idx = int(np.argmin(final_scores))
    return int(unvisited_nodes[idx])
```
